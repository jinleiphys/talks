import subprocess,re,math
S='/Users/jinlei/Desktop/code/FUSION/skills'
def calc(run):
    a,x=[],[]
    for ln in open('/tmp/fresco-runs/%s/fort.16'%run):
        m=re.match(r'\s*([\d.E+-]+)\s+([\d.E+-]+)\s*$',ln)
        if m: a.append(float(m.group(1))); x.append(float(m.group(2)))
    return a,x
def ip(a,x,t):
    for i in range(len(a)-1):
        if a[i]<=t<=a[i+1]:
            f=(t-a[i])/(a[i+1]-a[i]); return x[i]*(1-f)+x[i+1]*f
for run,sub,arg in [('n90zr-24','13160004',[]),('n90zr-55','22480005',['--energy','55','--columns','ANG-CM','DATA'])]:
    a,x=calc(run)
    out=subprocess.run(['python3',S+'/exfor-data/scripts/exfor.py','data',sub]+arg,capture_output=True,text=True).stdout
    r=[]
    for ln in out.splitlines():
        if ln.startswith('#'): continue
        p=ln.split()
        if len(p)>=2:
            try: t,d=float(p[0]),float(p[1])
            except ValueError: continue
            c=ip(a,x,t)
            if c: r.append(c/d)
    print('%s MeV: EXFOR %s, N=%d, mean calc/data=%.3f, rms(log10)=%.3f'%(run.split('-')[1],sub,len(r),sum(r)/len(r),math.sqrt(sum(math.log10(v)**2 for v in r)/len(r))))
PY
